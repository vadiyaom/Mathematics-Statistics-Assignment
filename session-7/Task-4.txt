4. Python Matplotlib Script (task4_spotify_marks.py):
--------------------------------------------------------------------------------
```python
import numpy as np
import matplotlib.pyplot as plt

# 1. Dataset
spotify_hours = [0.5, 1.0, 1.5, 2.0, 2.8, 3.5, 4.0, 4.8, 5.2, 6.0]
test_marks    = [95,  91,  86,  82,  75,  68,  60,  52,  45,  34 ]

# 2. Calculate Correlation
r = np.corrcoef(spotify_hours, test_marks)[0, 1]
print(f"Pearson Correlation Coefficient (r): {r:.4f}")

# 3. Scatter Plot with Trendline
plt.figure(figsize=(9, 6))
plt.scatter(spotify_hours, test_marks, color="#1DB954", s=120, edgecolors="black", label="Students")

# Trendline
m, c = np.polyfit(spotify_hours, test_marks, 1)
x_line = np.linspace(0.2, 6.3, 50)
plt.plot(x_line, m * x_line + c, color="#E74C3C", linestyle="--", label=f"Trendline: y = {m:.2f}x + {c:.2f}")

plt.title("Time Spent on Spotify vs. Test Marks (Strong Negative Correlation)")
plt.xlabel("Daily Time Spent on Spotify (Hours/Day)")
plt.ylabel("Test Marks (Out of 100)")
plt.grid(True, linestyle=":", alpha=0.6)
plt.legend()
plt.savefig("spotify_vs_marks_scatter.png")
plt.show()
```
